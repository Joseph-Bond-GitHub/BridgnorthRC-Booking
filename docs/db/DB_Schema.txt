
CREATE TYPE equipment_status_enum AS ENUM ('functional', 'under maintenance', 'retired');
CREATE TYPE equipment_category_enum AS ENUM ('boat', 'blade', 'gym');
CREATE TYPE boat_steering_mode_enum AS ENUM ('foot', 'hand', 'coxed');
CREATE TYPE rig_type AS ENUM ('sweep', 'scull');
CREATE TYPE member_access_level_enum AS ENUM ('user', 'admin', 'sysadmin');

-- type[] defines an array of values of that type 

CREATE TABLE equipment (
	equipment_id SMALLINT NOT NULL GENERATED ALWAYS AS IDENTITY CHECK (equipment_id >= 0),
	name VARCHAR(100) NOT NULL UNIQUE, 
	locked BOOLEAN NOT NULL, 
	additional_info VARCHAR(300), 
	status equipment_status_enum NOT NULL, 
	times_booked SMALLINT NOT NULL DEFAULT 0 CHECK (times_booked >= 0), 
	times_cancelled SMALLINT NOT NULL DEFAULT 0 CHECK (times_cancelled >= 0), 
	category equipment_category_enum NOT NULL, 
	PRIMARY KEY (equipment_id)
);

CREATE TABLE boat (
	equipment_id SMALLINT,
	steering_mode boat_steering_mode_enum[] NOT NULL CHECK((cardinality(steering_mode) = 1 AND steering_mode[1] IS NOT NULL) OR (cardinality(steering_mode) = 2 AND steering_mode[1] <> steering_mode[2] AND steering_mode[1] IS NOT NULL AND steering_mode[2] IS NOT NULL)),
	seats SMALLINT NOT NULL CHECK (seats IN(1,2,4,5,8)),
	average_crew_weight SMALLINT NOT NULL CHECK (average_crew_weight >= 0),
	boat_weight DECIMAL(5,2) CHECK (boat_weight >= 0), 
	last_rigged_as rig_type NOT NULL,
	PRIMARY KEY (equipment_id),
	FOREIGN KEY (equipment_id) REFERENCES equipment(equipment_id)
		ON DELETE CASCADE	
);

CREATE TABLE blade (
	equipment_id SMALLINT,
	blade_type rig_type NOT NULL,
	set_number SMALLINT NOT NULL,
	PRIMARY KEY (equipment_id),
	FOREIGN KEY (equipment_id) REFERENCES equipment(equipment_id)
		ON DELETE CASCADE
);

CREATE TABLE boat_blade (
	boat_equipment_id SMALLINT,
	blade_equipment_id SMALLINT,
	PRIMARY KEY (boat_equipment_id, blade_equipment_id),
	FOREIGN KEY (boat_equipment_id) REFERENCES boat(equipment_id)
		ON DELETE CASCADE,
	FOREIGN KEY (blade_equipment_id) REFERENCES blade(equipment_id)
		ON DELETE CASCADE
);

CREATE TABLE member (
	member_id SMALLINT NOT NULL GENERATED ALWAYS AS IDENTITY,
	name VARCHAR(100) NOT NULL,
	username VARCHAR(100) UNIQUE NOT NULL,
	password VARCHAR(255),
	access_level member_access_level_enum NOT NULL DEFAULT 'user',
	foot_steer BOOLEAN NOT NULL DEFAULT FALSE,
	tow BOOLEAN NOT NULL DEFAULT FALSE,
	is_active BOOLEAN NOT NULL DEFAULT TRUE,
	PRIMARY KEY (member_id)
);

CREATE TABLE booking (
	start_date_time TIMESTAMPTZ NOT NULL,
	end_date_time TIMESTAMPTZ NOT NULL CHECK (end_date_time > start_date_time),
	booking_id INTEGER GENERATED ALWAYS AS IDENTITY CHECK (booking_id >= 0),
	member_id SMALLINT NOT NULL,
	PRIMARY KEY (booking_id),
	FOREIGN KEY (member_id) REFERENCES member(member_id) 
		ON DELETE CASCADE
);

CREATE TABLE booking_equipment (
	booking_id INTEGER,
	equipment_id SMALLINT,
	PRIMARY KEY (booking_id, equipment_id),
	FOREIGN KEY (booking_id) REFERENCES booking(booking_id)
		ON DELETE CASCADE,
	FOREIGN KEY (equipment_id) REFERENCES equipment(equipment_id)
		ON DELETE CASCADE
);

CREATE INDEX number_booked_out_ind ON equipment (times_booked, times_cancelled);
CREATE INDEX which_equipment_indx ON equipment (category);
CREATE INDEX number_seats_indx ON boat (seats);
CREATE INDEX booking_times_indx ON booking (start_date_time, end_date_time);
CREATE INDEX equipment_lookup_indx ON booking_equipment (equipment_id);
CREATE INDEX numbers_blade_sets_indx ON blade (set_number);
